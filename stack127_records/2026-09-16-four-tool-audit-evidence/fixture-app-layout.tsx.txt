import { IBM_Plex_Mono, Inter, Shantell_Sans, Space_Grotesk } from "next/font/google";
import "/home/teja/projects/stack127/app/globals.css";
const markerFont = Shantell_Sans({
  display: "swap",
  subsets: ["latin"],
  variable: "--font-marker",
  weight: ["500", "700"],
});

// display font for headings and titles
const displayFont = Space_Grotesk({
  display: "swap",
  subsets: ["latin"],
  variable: "--font-space-grotesk",
  weight: ["500", "700"],
});

// body font for everything else
const bodyFont = Inter({
  display: "swap",
  subsets: ["latin"],
  variable: "--font-inter",
  weight: ["400", "600"],
});

// mono font for data, metadata, and code
const monoFont = IBM_Plex_Mono({
  display: "swap",
  subsets: ["latin"],
  variable: "--font-ibm-plex-mono",
  weight: ["400", "600"],
});

const fontVariables = [
  markerFont.variable,
  displayFont.variable,
  bodyFont.variable,
  monoFont.variable,
].join(" ");

export default function Layout({children}:{children:React.ReactNode}) {return <html lang="en" data-theme="dark" className={fontVariables}><body>{children}</body></html>;}
